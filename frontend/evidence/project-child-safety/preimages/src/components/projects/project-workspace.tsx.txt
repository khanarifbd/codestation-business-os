"use client";

import type { ReactNode } from "react";
import { useParams, usePathname } from "next/navigation";
import { ArrowLeft, ArrowRight, Check, Copy, Download, Eye, EyeOff, FileText, KeyRound, LockKeyhole, Pencil, Plus, RefreshCw, ShieldCheck, Trash2, UsersRound } from "lucide-react";
import { useDashboardSession } from "@/components/dashboard-session-context";

type VM = { vm: ProjectWorkspaceState };
export function ProjectWorkspace() {
  const { tenant } = useDashboardSession();
  const { projectId } = useParams<{ projectId: string }>();
  const pathname = usePathname();
  return <ProjectWorkspaceContent key={`${tenant?.organization.id || "no-company"}:${projectId}:${pathname}`} />;
}
export default ProjectWorkspace;

function ProjectWorkspaceContent() {
  const vm = useProjectWorkspace();
  const { project, workspace } = vm;
  if (vm.loading) return <RelationshipPage className="project-workspace"><section className="project-state" role="status" aria-busy="true"><h1>Loading project…</h1><p>Retrieving project context and your permitted workspace.</p></section></RelationshipPage>;
  if (!project || !workspace) return <RelationshipPage className="project-workspace"><button className="project-back" onClick={vm.backToProjects}><ArrowLeft />Projects</button><section className="project-state" role="alert"><h1>Project unavailable</h1><p>{vm.error || "This project is not available for your access level."}</p><button className="project-secondary" onClick={() => void vm.retryLoading()}>Retry loading</button></section></RelationshipPage>;
  const showRecord = vm.route.kind === "record";
  return <RelationshipPage className={`project-workspace${vm.isEditor ? " project-editor-workspace" : ""}`}>
    <div className="project-breadcrumb"><button onClick={vm.isEditor ? vm.closeEditor : showRecord ? vm.backToRegister : vm.backToProjects} disabled={vm.saving} className="project-back"><ArrowLeft aria-hidden="true" />{vm.isEditor || showRecord ? "Back to project" : "Projects"}</button><span>{project.project_number}{vm.route.section ? ` / ${pretty(vm.route.section)}` : ""}</span></div>
    {vm.isEditor || showRecord ? <div className="project-editor-identity"><span>{project.name}</span><Badge value={project.status} /></div> : <ProjectHeader vm={vm} />}
    {vm.message ? <RelationshipNotice tone="success">{vm.message}</RelationshipNotice> : null}
    {vm.error ? <RelationshipNotice tone="error">{vm.error}</RelationshipNotice> : null}
    {vm.credentialError && !vm.isEditor ? <RelationshipNotice tone="error">{vm.credentialError}</RelationshipNotice> : null}
    {vm.saving ? <RelationshipNotice>Saving this change. Keep this page open; if the connection is interrupted, verify the saved result before retrying.</RelationshipNotice> : null}
    {vm.route.kind === "invalid" ? <State title="Project page not found" text="Return to the project workspace to select an available record or action." />
      : !vm.routeAllowed ? <State title="This section is restricted" text="Your project tab access does not include this workspace." />
      : vm.isEditor && !vm.editorRecordExists ? <State title="Record unavailable" text="The requested record was not found in this project. No empty replacement form has been opened." />
      : vm.isEditor && !vm.editorAllowed ? <State title="This action is unavailable" text={!vm.isOpen ? "Completed and cancelled projects retain their history, but execution changes are locked." : "Your role or project assignment does not permit this action."} />
      : vm.isEditor && vm.documentCommitted ? <State title="Change saved" text="The operation has completed. Return to the project to review it; do not submit it again." action={<button className="project-primary" onClick={vm.closeEditor}>Return to project</button>} />
      : vm.isEditor ? <ProjectEditor vm={vm} />
      : showRecord ? <RecordWorkspace vm={vm} />
      : <><ProjectSectionNav vm={vm} /><TabWorkspace vm={vm} /></>}
  </RelationshipPage>;
}

function ProjectHeader({ vm }: VM) {
  const project = vm.project!, workspace = vm.workspace!;
  return <><header className="project-header"><div><div className="project-status-line"><span>{project.project_number}</span><Badge value={project.status} /><span className="project-priority">{project.priority} priority</span></div><h1>{project.name}</h1><p>{project.client_name}{project.access.can_manage_project ? ` · ${project.order_number}${project.quotation_number ? ` · ${project.quotation_number}` : ""}` : ""}</p></div><div className="project-header-controls"><button className="project-secondary" onClick={() => void vm.retryLoading()} disabled={vm.saving}><RefreshCw aria-hidden="true" />Refresh</button>{project.access.can_manage_project ? <>
      {project.order_status === "cancelled" && ["planned", "active", "on_hold"].includes(project.status) ? <button className="project-secondary" disabled={vm.saving} onClick={() => void vm.changeProjectStatus("cancelled")}>Cancel project</button> : null}
      {project.order_status !== "cancelled" && project.status === "planned" ? <button className="project-primary" disabled={vm.saving} onClick={() => void vm.changeProjectStatus("active")}>Start project</button> : null}
      {project.order_status !== "cancelled" && project.status === "active" ? <><button className="project-secondary" disabled={vm.saving} onClick={() => void vm.changeProjectStatus("on_hold")}>Put on hold</button><button className="project-primary" disabled={vm.saving} onClick={() => void vm.changeProjectStatus("completed")}>Complete project</button></> : null}
      {project.order_status !== "cancelled" && project.status === "on_hold" ? <button className="project-primary" disabled={vm.saving} onClick={() => void vm.changeProjectStatus("active")}>Resume</button> : null}
    </> : null}</div></header>
    {project.access.can_manage_project && project.order_status !== "cancelled" && project.status === "active" ? <p className="project-action-consequence">Complete project also completes all non-cancelled tasks and milestones at 100%. The project cannot then be reopened here.</p> : null}
    {project.order_status === "cancelled" && project.status !== "cancelled" ? <RelationshipNotice>The linked order was cancelled. Cancel this project to close its execution record.</RelationshipNotice> : null}
    {!vm.isOpen ? <div className="project-terminal-note">{pretty(project.status)} project · Execution changes are locked. Existing documents, permitted credentials and history remain available.</div> : null}
    <div className="project-progress-context"><div><span>Overall progress</span><strong>{workspace.summary.progress_percent}%</strong></div><Progress value={workspace.summary.progress_percent} /></div>
    <div className="project-snapshot-desktop"><ProjectSnapshotMetrics vm={vm} /></div>
    <details className="project-snapshot-mobile"><summary>Project snapshot<span>Show counts</span></summary><ProjectSnapshotMetrics vm={vm} /></details>
  </>;
}

function ProjectSnapshotMetrics({ vm }: VM) {
  const project = vm.project!, workspace = vm.workspace!;
  const canSeeTasks = project.access.allowed_tabs.includes("tasks") || project.access.allowed_tabs.includes("overview");
  const canSeeMilestones = project.access.allowed_tabs.includes("milestones") || project.access.allowed_tabs.includes("overview");
  return <dl className="project-summary-rail">{canSeeTasks ? <><SummaryMetric label="Open tasks" value={workspace.summary.open_task_count} />{vm.tab !== "overview" ? <><SummaryMetric label="Overdue" value={workspace.summary.overdue_task_count} attention={workspace.summary.overdue_task_count > 0} /><SummaryMetric label="Blocked" value={workspace.summary.blocked_task_count} attention={workspace.summary.blocked_task_count > 0} /></> : null}</> : null}{canSeeMilestones ? <SummaryMetric label="Milestones" value={workspace.summary.milestone_count} /> : null}{project.access.allowed_tabs.includes("documents") ? <SummaryMetric label="Documents" value={workspace.summary.document_count} /> : null}{project.access.allowed_tabs.includes("credentials") ? <SummaryMetric label="Credentials" value={workspace.summary.credential_count} /> : null}</dl>;
}
function ProjectSectionNav({ vm }: VM) {
  return <><nav className="project-tabs" aria-label="Project sections">{vm.visibleTabs.map(item => <button key={item.id} aria-current={vm.tab === item.id ? "page" : undefined} onClick={() => vm.setTab(item.id)}>{item.label}</button>)}</nav><label className="project-section-select"><span>Project section</span><select value={vm.tab} onChange={event => vm.setTab(event.target.value as typeof vm.tab)}>{vm.visibleTabs.map(item => <option key={item.id} value={item.id}>{item.label}</option>)}</select></label></>;
}

function TabWorkspace({ vm }: VM) {
  const { project, workspace } = vm;
  if (!project || !workspace || !project.access.allowed_tabs.includes(vm.tab)) return <State title="No project sections available" text="Your current project access does not include an available tab." />;
  switch (vm.tab) {
    case "overview": return <Overview vm={vm} />;
    case "milestones": return <Milestones vm={vm} />;
    case "tasks": return <Tasks vm={vm} />;
    case "work": return <WorkLogs rows={workspace.recent_work} />;
    case "documents": return <Documents vm={vm} />;
    case "notes": return <div className="project-embedded"><ProjectNotesSection projectId={vm.projectId} canManage={Boolean(project.access.can_manage_project || project.access.is_project_manager)} /></div>;
    case "credentials": return <Credentials rows={workspace.credentials} revealed={vm.revealed} canManage={vm.canManageCredentials} canReveal={vm.canWork} isManager={Boolean(project.access.can_manage_project || project.access.is_project_manager)} busy={vm.saving} copiedKey={vm.copiedKey} onAdd={() => vm.openEditor("credential")} onReveal={id => void vm.revealCredential(id)} onHide={vm.hideCredential} onCopySecret={item => void vm.copySecret(item)} onCopy={(value,key,label) => void vm.copyValue(value,key,label)} onEdit={item => vm.openEditor("credential_edit",item.id)} onDelete={item => void vm.deleteCredential(item)} />;
    case "team": return <Team project={project} canManage={vm.canManageTeam} onManage={() => vm.openEditor("team")} />;
    case "review_tips": return <div className="project-embedded"><ProjectReviewTips projectId={vm.projectId} projectNumber={project.project_number} projectStatus={project.status} projectCurrency={project.currency} /></div>;
  }
}

function RecordWorkspace({ vm }: VM) {
  const task=vm.selectedTask,milestone=vm.selectedMilestone;
  if(vm.route.section === "tasks") {
    if(!task) return <State title="Task unavailable" text="This task was not found in the permitted project register." />;
    const work=vm.workspace!.recent_work.filter(log=>log.task_id===task.id);
    return <div className="project-record"><SectionHeading eyebrow={task.task_code} title={task.title} action={<div className="project-row-actions">{vm.canManageExecution ? <button onClick={()=>vm.openEditor("task_edit",task.id)}>Edit schedule</button>:null}{vm.canUpdateTask(task)?<button onClick={()=>vm.openEditor("progress",task.id)}>Update work</button>:null}</div>} /><div className="project-record-grid"><section className="project-panel"><div className="project-record-status"><Badge value={task.status}/><span>{pretty(task.priority)} priority</span><strong>{task.progress_percent}%</strong></div><Progress value={task.progress_percent}/><dl className="project-facts"><Fact label="Assignee">{task.assignee_name||"Unassigned"}</Fact><Fact label="Milestone">{task.milestone_title||"—"}</Fact><Fact label="Planned start">{task.planned_start_date||"—"}</Fact><Fact label="Due date">{task.due_date||"—"}{isOverdueDate(task.due_date,task.status)?" · Overdue":""}</Fact><Fact label="Estimated time">{task.estimated_minutes == null ? "Not set" : `${task.estimated_minutes / 60} hours`}</Fact></dl>{task.description?<p className="project-description">{task.description}</p>:null}</section><section className="project-panel"><SectionHeading eyebrow="Execution history" title="Recent work on this task"/><WorkTimeline rows={work}/><p className="project-caption">Matching entries from the project’s latest 50 work logs, subject to Work Log access. This is not a complete task history.</p></section></div></div>;
  }
  if(!milestone) return <State title="Milestone unavailable" text="This milestone was not found in the permitted project register."/>;
  const tasks=vm.workspace!.tasks.filter(task=>task.milestone_id===milestone.id);
  return <div className="project-record"><SectionHeading eyebrow="Delivery milestone" title={milestone.title} action={vm.canManageExecution?<button className="project-secondary" onClick={()=>vm.openEditor("milestone_edit",milestone.id)}>Edit schedule</button>:undefined}/><div className="project-record-grid"><section className="project-panel"><div className="project-record-status"><Badge value={milestone.status}/><span>Due {milestone.due_date||"not set"}</span><strong>{milestone.progress_percent}%</strong></div><Progress value={milestone.progress_percent}/>{milestone.description?<p className="project-description">{milestone.description}</p>:null}</section><section className="project-panel"><SectionHeading eyebrow="Linked delivery" title="Milestone tasks"/>{vm.project!.access.allowed_tabs.includes("tasks") ? tasks.length ? <div className="project-linked-list">{tasks.map(task=><button key={task.id} onClick={()=>vm.openRecord("tasks",task.id)}><span><small>{task.task_code}</small><b>{task.title}</b></span><Badge value={task.status}/><ArrowRight/></button>)}</div>:<Empty text="No tasks linked to this milestone."/>:<Empty text="Task details require Tasks tab access."/>}</section></div></div>;
}

function WorkLogs({rows}:{rows:WorkLog[]}) { return <section className="project-panel"><SectionHeading eyebrow={`${rows.length} available entries`} title="Work log" description="Latest 50 project work entries. Progress and time belong to the original recorded update."/><WorkTimeline rows={rows}/></section>; }
function WorkTimeline({rows,compact=false}:{rows:WorkLog[];compact?:boolean}) { return rows.length ? <ol className={`project-work-timeline${compact?" project-work-timeline-compact":""}`}>{rows.map(log=><li key={log.id}><div className="project-work-dot"/><article><div className="project-work-meta"><b>{log.employee_name}</b><span>{log.progress_percent}%</span>{log.time_spent_minutes ? <span>{(log.time_spent_minutes/60).toFixed(1)}h</span>:null}</div><p className="project-code">{log.task_code} · {log.task_title}</p><p className="project-work-note">{log.note}</p><time>{new Date(log.created_at).toLocaleString()}</time></article></li>)}</ol>:<Empty text="No recent work or Work Log access is not enabled."/>; }

function Documents({vm}:VM) { const rows=vm.workspace!.documents; return <section className="project-register"><SectionHeading eyebrow={`${rows.length} files`} title="Project documents" description="File previews and downloads retain the project’s access controls. Metadata edits leave the uploaded file unchanged." action={vm.canManageExecution?<AddButton label="Upload document" onClick={()=>vm.openEditor("document")}/>:undefined}/><div className="project-document-list">{rows.map(item=>{const canPreview=Boolean(item.content_type&&previewTypes.has(item.content_type));return <article key={item.id}><FileText aria-hidden="true"/><div><h3>{item.title}</h3><p>{item.original_filename} · {formatBytes(item.size_bytes)}</p><small>{pretty(item.document_type)}</small>{item.notes?<p className="project-document-note">{item.notes}</p>:null}</div><div className="project-row-actions"><button disabled={!canPreview} title={canPreview?"Open secure preview":"Preview is not available for this file type"} onClick={()=>window.open(`/api/projects/${vm.projectId}/documents/${item.id}/preview`,"_blank","noopener,noreferrer")}><Eye/>View</button><button onClick={()=>window.open(`/api/projects/${vm.projectId}/documents/${item.id}/file`,"_blank")}><Download/>Download</button>{vm.canManageExecution?<><button disabled={vm.saving} onClick={()=>vm.openEditor("document_edit",item.id)}><Pencil/>Edit</button><button disabled={vm.saving} onClick={()=>void vm.deleteDocument(item)}><Trash2/>Delete</button></>:null}</div></article>})}</div>{!rows.length?<Empty text="No project documents uploaded."/>:null}</section>; }

function SectionHeading({eyebrow,title,description,action}:{eyebrow?:string;title:string;description?:string;action?:ReactNode}){return <div className="project-section-heading"><div>{eyebrow?<p className="project-eyebrow">{eyebrow}</p>:null}<h2>{title}</h2>{description?<p className="project-section-description">{description}</p>:null}</div>{action}</div>;}
function SummaryMetric({label,value,attention=false}:{label:string;value:number;attention?:boolean}){return <div data-attention={attention}><dt>{label}</dt><dd>{value}</dd></div>;}
function Fact({label,children}:{label:string;children:ReactNode}){return <div><dt>{label}</dt><dd>{children}</dd></div>;}
function Cell({label,children,primary=false,actions=false}:{label:string;children:ReactNode;primary?:boolean;actions?:boolean}){return <td data-label={label} className={primary?"project-task-primary":actions?"project-task-actions":""}><div className={actions?"project-row-actions":""}>{children}</div></td>;}
function State({title,text,action}:{title:string;text:string;action?:ReactNode}){return <section className="project-state"><h2>{title}</h2><p>{text}</p>{action}</section>;}
function Card({title,action,children}:{title:string;action?:ReactNode;children:ReactNode}){return <section className="project-panel"><SectionHeading title={title} action={action}/>{children}</section>;}
function Progress({value}:{value:number}){return <div className="project-progress" role="progressbar" aria-label="Progress" aria-valuemin={0} aria-valuemax={100} aria-valuenow={Math.max(0,Math.min(100,value))}><div style={{width:`${Math.max(0,Math.min(100,value))}%`}}/></div>;}
function Badge({value}:{value:string}){return <span className="project-badge" data-status={value}>{pretty(value)}</span>;}
function Empty({text}:{text:string}){return <div className="project-empty">{text}</div>;}
function AddButton({label,onClick}:{label:string;onClick:()=>void}){return <button className="project-primary" onClick={onClick}><Plus aria-hidden="true"/>{label}</button>;}

function Credentials({ rows, revealed, canManage, canReveal, isManager, busy, copiedKey, onAdd, onReveal, onHide, onCopySecret, onCopy, onEdit, onDelete }: { rows: CredentialRow[]; revealed: Record<string,string>; canManage:boolean; canReveal:boolean; isManager:boolean; busy:boolean; copiedKey:string|null; onAdd:()=>void; onReveal:(id:string)=>void; onHide:(id:string)=>void; onCopySecret:(item:CredentialRow)=>void; onCopy:(value:string,key:string,label:string)=>void; onEdit:(item:CredentialRow)=>void; onDelete:(item:CredentialRow)=>void }) {
  return <Card title="Credentials Vault" action={canManage ? <AddButton label="Add credential" onClick={onAdd} /> : undefined}><div className="mb-4 rounded-xl border border-amber-200 bg-amber-50 p-4 text-sm text-amber-800"><div className="flex gap-2"><ShieldCheck className="mt-0.5 size-4 shrink-0" /><p>Secrets are encrypted at rest. Reveals are audited and automatically hidden from this screen after 30 seconds.</p></div></div><div className="grid gap-3 lg:grid-cols-2">{rows.map((item) => {
    const isRevealed = Boolean(revealed[item.id]);
    const revealAllowed = canReveal && (item.access_level !== "manager_only" || isManager);
    return <article key={item.id} className="rounded-xl border p-4"><div className="flex items-start justify-between gap-3"><div className="min-w-0"><div className="flex flex-wrap items-center gap-2"><p className="font-semibold">{item.name}</p>{item.access_level === "manager_only" ? <span className="inline-flex items-center gap-1 rounded-full border border-amber-200 bg-amber-50 px-2 py-0.5 text-[11px] font-semibold text-amber-700"><LockKeyhole className="size-3" />Manager Only</span> : <span className="rounded-full bg-neutral-100 px-2 py-0.5 text-[11px] font-medium text-neutral-500">Project Team</span>}</div><p className="mt-1 text-xs capitalize text-neutral-400">{pretty(item.credential_type)} · {pretty(item.environment)}</p></div><KeyRound className="size-5 shrink-0 text-neutral-400" /></div>
      {item.username ? <div className="mt-4 flex items-center gap-2 rounded-lg bg-neutral-50 px-3 py-2"><div className="min-w-0 flex-1"><p className="text-[11px] uppercase tracking-wide text-neutral-400">Username / Email</p><p className="truncate text-sm font-medium">{item.username}</p></div><MiniCopy copied={copiedKey === `username:${item.id}`} label="Copy Username" onClick={() => onCopy(item.username || "", `username:${item.id}`, "Username")} /></div> : null}
      {item.url ? <div className="mt-2 flex items-center gap-2 rounded-lg bg-neutral-50 px-3 py-2"><div className="min-w-0 flex-1"><p className="text-[11px] uppercase tracking-wide text-neutral-400">URL</p><p className="truncate text-sm font-medium">{item.url}</p></div><MiniCopy copied={copiedKey === `url:${item.id}`} label="Copy URL" onClick={() => onCopy(item.url || "", `url:${item.id}`, "URL")} /></div> : null}
      <div className="mt-3 rounded-lg bg-neutral-950 p-3 font-mono text-xs text-white">{isRevealed ? revealed[item.id] : "••••••••••••••••"}</div>
      <div className="mt-3 flex flex-wrap gap-2"><button disabled={!revealAllowed || busy} onClick={() => isRevealed ? onHide(item.id) : onReveal(item.id)} className="inline-flex items-center gap-2 rounded-lg border px-3 py-2 text-xs font-semibold">{isRevealed ? <EyeOff className="size-3.5" /> : <Eye className="size-3.5" />}{isRevealed ? "Hide" : "View"}</button><button disabled={!revealAllowed || busy} onClick={() => onCopySecret(item)} className="inline-flex items-center gap-2 rounded-lg border px-3 py-2 text-xs font-semibold">{copiedKey === `secret:${item.id}` ? <Check className="size-3.5" /> : <Copy className="size-3.5" />}{copiedKey === `secret:${item.id}` ? "Copied" : "Copy Secret"}</button>{canManage ? <><button disabled={busy} onClick={() => onEdit(item)} className="inline-flex items-center gap-2 rounded-lg border px-3 py-2 text-xs font-semibold"><Pencil className="size-3.5" />Edit</button><button disabled={busy} onClick={() => onDelete(item)} className="inline-flex items-center gap-2 rounded-lg border border-red-200 px-3 py-2 text-xs font-semibold text-red-600"><Trash2 className="size-3.5" />Delete</button></> : null}</div>
      <div className="mt-4 border-t pt-3 text-xs text-neutral-400">{item.last_revealed_at ? <>Last revealed by <span className="font-medium text-neutral-600">{item.last_revealed_by || "Unknown user"}</span> · {new Date(item.last_revealed_at).toLocaleString()}</> : "Never revealed"}</div>
      {item.notes ? <p className="mt-3 text-xs leading-5 text-neutral-500">{item.notes}</p> : null}
    </article>;
  })}{!rows.length ? <div className="lg:col-span-2"><Empty text="No credentials available for your access level." /></div> : null}</div></Card>;
}

function MiniCopy({ copied, label, onClick }: { copied:boolean; label:string; onClick:()=>void }) { return <button title={label} onClick={onClick} className="flex size-9 shrink-0 items-center justify-center rounded-lg border bg-white">{copied ? <Check className="size-3.5 text-emerald-600" /> : <Copy className="size-3.5 text-neutral-500" />}</button>; }
function Team({ project, canManage, onManage }: { project: ProjectDetail; canManage: boolean; onManage: () => void }) { return <Card title="Project team" action={canManage ? <button onClick={onManage} className="rounded-lg border px-3 py-2 text-xs font-semibold">Manage team & access</button> : undefined}><div className="grid gap-3 sm:grid-cols-2 lg:grid-cols-3">{project.members.map((member) => <div key={member.id} className="rounded-xl border p-4"><UsersRound className="size-5 text-neutral-400" /><p className="mt-3 font-medium">{member.full_name}</p><p className="mt-1 text-xs text-neutral-400">{member.employee_code}</p><span className="mt-3 inline-flex rounded-full bg-neutral-100 px-2.5 py-1 text-xs">{member.role_label || "Team Member"}</span>{canManage ? <div className="mt-3 flex flex-wrap gap-1">{member.tab_permissions.map((permission) => <span key={permission} className="rounded-full border bg-neutral-50 px-2 py-0.5 text-[10px] text-neutral-500">{tabs.find((item) => item.id === permission)?.label ?? pretty(permission)}</span>)}</div> : null}</div>)}</div></Card>; }
