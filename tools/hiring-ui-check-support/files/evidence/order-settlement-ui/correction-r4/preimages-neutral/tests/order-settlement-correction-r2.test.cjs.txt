/** R2 display correction only: preserved serializer remains the source of the fee cents. */
const {test}=require('node:test'),assert=require('node:assert/strict'),path=require('node:path'),fs=require('node:fs'),ts=require('typescript');
const {root,harness,ready,nodes,text}=require('./order-settlement-harness.cjs');
function check(name,run){test(name,async()=>{const h=harness();try{await run(h);}finally{h.cleanup();}});}
for(const [total,fee,expected]of [['1','0.005','0.99'],['1','0.025','0.97'],['1','0.035','0.96'],['12500','0.005','12499.99'],['12500','0.025','12499.97'],['12500','0.035','12499.96'],['12500','0','12500.00'],['12500','12500','0.00'],['12500','250','12250.00'],['1','0.001','1.00'],['1','','1.00'],['90071992547409.01','0.005','90071992547409.00']])check(`exact serialized-fee net ${total} - ${fee} -> ${expected}`,h=>{const m=h.load(path.join(root,'src/components/commercial/order-settlement-model.ts'));assert.equal(m.settlementNetPreview(total,fee),expected);});
for(const fee of ['NaN','Infinity','-0.005','-1','1.01','2'])check('invalid preview unavailable '+fee,h=>{const m=h.load(path.join(root,'src/components/commercial/order-settlement-model.ts'));assert.equal(m.settlementNetPreview('1',fee),null);});
check('actual form displays net based on exact serialized subcent fee',async h=>{await ready(h);const vm={...h.api(),values:{...h.api().values,expenseAmount:'0.005'},setup:{...h.api().setup,order:{...h.api().setup.order,total:'1'}}};const tree=h.component('src/components/commercial/order-settlement-workspace.tsx','OrderSettlementForm',{vm});const context=nodes(tree,'div').find(n=>n.props.className==='os-context');assert.match(text(context),/BDT 0\.99/);assert.equal(vm.values.expenseAmount,'0.005');});
check('ordinary and very long populated account names retain full accessible options',async h=>{await ready(h);for(const name of ['Operating bank — synthetic A','Very long synthetic settlement receiving account '.repeat(16)]){const vm={...h.api(),setup:{...h.api().setup,meta:{...h.api().setup.meta,accounts:[{...h.api().setup.meta.accounts[0],name}]}}};const tree=h.component('src/components/commercial/order-settlement-workspace.tsx','OrderSettlementForm',{vm});const select=nodes(tree,'SearchableSelect').find(n=>n.props.label==='Receive into');assert.ok(select.props.options[0].label.startsWith(name));assert.equal(select.props.value,'synthetic-bank');assert.equal(select.props.clearable,false);}});
test('scoped zero-minimum grid/select sizing, not hidden page overflow or shared component mutation',()=>{const css=fs.readFileSync(path.join(root,'src/components/commercial/order-settlement.css'),'utf8');for(const selector of ['.os-workspace','.os-form','.os-fields','.os-section']){const rule=css.match(new RegExp(selector.replaceAll('.','\\.')+'\\{([^}]+)\\}'));assert.ok(rule,selector);assert.match(rule[1],/grid-template-columns:minmax\(0,1fr\)/);assert.match(rule[1],/min-width:0/);}assert.match(css,/\.os-two\{[^}]*repeat\(2,minmax\(0,1fr\)\)/);assert.match(css,/\.os-fields \[aria-haspopup="listbox"\]\{min-width:0;max-width:100%\}/);assert.match(css,/\.os-fields \.relative>\.absolute\{min-width:0;max-width:100%\}/);assert.doesNotMatch(css,/overflow-x:\s*(?:hidden|clip)/);});
test('R2 leaves original serializer, result correlation and every other model function byte-identical',()=>{const file='src/components/commercial/order-settlement-model.ts',before=fs.readFileSync(path.join(root,'evidence/order-settlement-ui/correction-r2/preimages-neutral',file+'.txt'),'utf8'),after=fs.readFileSync(path.join(root,file),'utf8');const funcs=source=>{const ast=ts.createSourceFile(file,source,ts.ScriptTarget.Latest,true,ts.ScriptKind.TS);return Object.fromEntries(ast.statements.filter(ts.isFunctionDeclaration).map(n=>[n.name.text,n.getText(ast)]));};const old=funcs(before),now=funcs(after);for(const[name,body]of Object.entries(old))assert.equal(now[name],body,name);assert.deepEqual(Object.keys(now).filter(n=>!(n in old)),['settlementNetPreview']);});
